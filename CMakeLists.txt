cmake_minimum_required(VERSION 3.20)
project(CircuitForge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CIRCUITFORGE_CUDA "Build CUDA backend" OFF)
add_library(CircuitCore src/NetlistParser.cpp src/CircuitCompiler.cpp src/PatternGenerator.cpp src/CpuSimulator.cpp src/FaultGenerator.cpp src/FaultCollapser.cpp src/AtpgEngine.cpp src/Benchmark.cpp)
target_include_directories(CircuitCore PUBLIC include)
add_executable(CircuitForge src/Main.cpp)
target_link_libraries(CircuitForge PRIVATE CircuitCore)
enable_testing()
add_executable(CircuitTests tests/TestCore.cpp)
target_link_libraries(CircuitTests PRIVATE CircuitCore)
add_test(NAME Core COMMAND CircuitTests)
set_tests_properties(Core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
if(CIRCUITFORGE_CUDA)
 enable_language(CUDA)
 target_sources(CircuitCore PRIVATE src/GpuSimulator.cu)
 target_compile_definitions(CircuitCore PUBLIC CIRCUITFORGE_CUDA)
 set_target_properties(CircuitCore PROPERTIES CUDA_ARCHITECTURES 75 CUDA_STANDARD 17)
 target_compile_options(CircuitCore PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ptxas-options=-v>)
endif()

add_test(NAME CliSimulate COMMAND CircuitForge simulate "${CMAKE_CURRENT_SOURCE_DIR}/circuits/FullAdder.txt")
add_test(NAME CliBenchImport COMMAND CircuitForge faults "${CMAKE_CURRENT_SOURCE_DIR}/circuits/benchmarks/Smoke.bench")
add_test(NAME CliAtpg COMMAND CircuitForge atpg "${CMAKE_CURRENT_SOURCE_DIR}/circuits/FullAdder.txt")
add_test(NAME CliCollapse COMMAND CircuitForge collapse "${CMAKE_CURRENT_SOURCE_DIR}/circuits/FullAdder.txt")
add_test(NAME CliInvalidOption COMMAND CircuitForge faults "${CMAKE_CURRENT_SOURCE_DIR}/circuits/FullAdder.txt" --patterns -1)
add_test(NAME CliInvalidBlock COMMAND CircuitForge faults "${CMAKE_CURRENT_SOURCE_DIR}/circuits/FullAdder.txt" --block-size 7)
add_test(NAME CliInvalidCommand COMMAND CircuitForge unknown "${CMAKE_CURRENT_SOURCE_DIR}/circuits/FullAdder.txt")
set_tests_properties(CliInvalidOption CliInvalidBlock CliInvalidCommand PROPERTIES WILL_FAIL TRUE)
if(CIRCUITFORGE_CUDA)
 add_test(NAME CliGpuDropping COMMAND CircuitForge faults "${CMAKE_CURRENT_SOURCE_DIR}/circuits/FullAdder.txt" --gpu --patterns 65 --fault-dropping)
endif()
